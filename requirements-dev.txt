# Type-checking dependencies for the Python plotting modules under cpp/ and the
# Python demo suites under Python/ (scope and mypy_path are defined in mypy.ini).
#
#   .venv/bin/pip install -r requirements-dev.txt
#
# These are NOT needed to generate figures -- the plotters run on matplotlib,
# pandas and plotly alone (see the hint in each demo's generate_plots.sh).
# They are only needed to run the linter, configured in mypy.ini.
# Launch from the repository root so the relative mypy_path entries resolve:
#
#   mypy cpp/trex_selector_methods/ad/trex_core/trex_plt_utils.py \
#        cpp/trex_selector_methods/ad/trex_da/trex_da_plt_utils.py
#   mypy Python/tsolvers/ad/demo_ts_01_tlars/demo_ts_01_tlars.py

mypy

# pandas ships no py.typed marker, so without this the checker cannot see pandas
# at all and reports 'Library stubs not installed for "pandas"'. The stub
# releases track pandas version-for-version, so keep this matched to the pandas
# in the venv (pandas-stubs 3.0.3.* <-> pandas 3.0.3).
pandas-stubs~=3.0.3
